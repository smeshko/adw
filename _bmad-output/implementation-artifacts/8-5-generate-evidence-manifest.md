**Pydantic Models:**

```python
from pydantic import BaseModel, Field
from datetime import datetime
from enum import Enum
from pathlib import Path

class EvidenceType(str, Enum):
    CLI = "cli"
    SCREENSHOT = "screenshot"
    API = "api"
    LOG = "log"

class EvidenceStatus(str, Enum):
    PASS = "pass"
    FAIL = "fail"
    ERROR = "error"
    SKIPPED = "skipped"

class EvidenceItem(BaseModel):
    """Single piece of captured evidence."""
    name: str
    type: EvidenceType
    path: str  # Relative path within evidence directory
    status: EvidenceStatus
    plan_step: str | None = None  # Reference to plan.md step (e.g., "step_3")
    details: dict | None = None   # Type-specific details

class PlanStepCoverage(BaseModel):
    """Coverage information for a single plan step."""
    step_id: str
    step_description: str | None = None
    evidence_items: list[str] = Field(default_factory=list)  # Names of linked evidence
    covered: bool = False

class CoverageSummary(BaseModel):
    """Summary of evidence coverage."""
    total_plan_steps: int
    covered_steps: int
    uncovered_steps: int
    coverage_percentage: float
    uncovered_step_ids: list[str]

class EvidenceManifest(BaseModel):
    """Complete evidence manifest for a run."""
    run_id: str
    generated_at: datetime = Field(default_factory=datetime.now)
    platform: str
    evidence_directory: str

    # Summary statistics
    total_items: int
    passed: int
    failed: int
    errors: int
    skipped: int

    # Evidence items
    items: list[EvidenceItem]

    # Plan coverage (optional - only if plan.md exists)
    coverage: CoverageSummary | None = None
    step_coverage: list[PlanStepCoverage] | None = None
```

**Plan Step Parsing:**
```python
import re
from pathlib import Path

def parse_plan_steps(plan_path: Path) -> list[tuple[str, str]]:
    """Extract step IDs and descriptions from plan.md.

    Returns list of (step_id, description) tuples.
    """
    steps = []
    content = plan_path.read_text()

    # Match patterns like "## Step 1: Description" or "### 1.1 Description"
    patterns = [
        r'^##\s+Step\s+(\d+):\s*(.+)$',
        r'^###\s+(\d+\.\d+)\s+(.+)$',
        r'^-\s+\[.\]\s+(.+)$',  # Checkbox items
    ]

    for line in content.split('\n'):
        for pattern in patterns:
            match = re.match(pattern, line.strip())
            if match:
                if len(match.groups()) == 2:
                    steps.append((f"step_{match.group(1)}", match.group(2)))
                else:
                    steps.append((f"task_{len(steps)}", match.group(1)))
                break

    return steps
```

**Evidence Linking Heuristics:**
```python
def link_evidence_to_step(evidence_name: str, plan_steps: list[tuple[str, str]]) -> str | None:
    """Attempt to link evidence item to a plan step.

    Uses keyword matching between evidence name and step descriptions.
    """
    evidence_lower = evidence_name.lower()

    for step_id, description in plan_steps:
        desc_lower = description.lower()

        # Exact match on key terms
        if evidence_lower in desc_lower or desc_lower in evidence_lower:
            return step_id

        # Keyword overlap
        evidence_words = set(evidence_lower.split('_'))
        desc_words = set(desc_lower.split())
        if evidence_words & desc_words:  # Non-empty intersection
            return step_id

    return None
```

```python
def calculate_coverage(
    total_steps: int,
    covered_steps: int,
) -> float:
    if total_steps == 0:
        return 100.0  # No steps = fully covered by definition
    return round((covered_steps / total_steps) * 100, 1)
```
